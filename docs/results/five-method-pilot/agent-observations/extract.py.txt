"""Fit mixture HMMs to permitted cached transformer observations; no model calls.
Run with the supplied virtualenv Python. Only reads/writes this directory.
"""
from pathlib import Path
import argparse, json, time
import numpy as np
import torch

ROOT=Path(__file__).resolve().parent

def kmeans(x,k,rng,iters=60):
    c=x[rng.choice(len(x), k, replace=False)].copy()
    for _ in range(iters):
        label=((x[:,None]-c[None])**2).sum(-1).argmin(1)
        new=np.array([x[label==j].mean(0) if np.any(label==j) else x[rng.integers(len(x))] for j in range(k)])
        if np.max(np.abs(new-c))<1e-8: break
        c=new
    return c,label

class Fits(torch.nn.Module):
    def __init__(self,x,p,K,S,R,seed):
        super().__init__(); rng=np.random.default_rng(seed); V=p.shape[-1]
        cent,labels=kmeans(p[:,8:].mean(1),K,rng)
        E=[]; A=[]; I=[]; W=[]
        for r in range(R):
            if r<R*3//4:
                # Each component begins near an unsupervised late-history cluster.
                ee=np.log(np.maximum(cent[:,None,:],1e-6))+rng.normal(0,0.8,(K,S,V))
                ww=np.log(np.bincount(labels,minlength=K)+1)
            else:
                ee=np.log(rng.dirichlet(np.ones(V)*0.8,size=(K,S)))
                ww=rng.normal(0,.2,K)
            E.append(ee); A.append(rng.normal(0,.65,(K,S,S))); I.append(rng.normal(0,.65,(K,S))); W.append(ww)
        self.el=torch.nn.Parameter(torch.tensor(np.array(E),dtype=torch.float32))
        self.al=torch.nn.Parameter(torch.tensor(np.array(A),dtype=torch.float32))
        self.il=torch.nn.Parameter(torch.tensor(np.array(I),dtype=torch.float32))
        self.wl=torch.nn.Parameter(torch.tensor(np.array(W),dtype=torch.float32))
    def params(self):
        return self.wl.softmax(-1),self.il.softmax(-1),self.al.softmax(-1),self.el.softmax(-1)
    def score(self,x,target,detail=False):
        w,i,a,e=self.params(); R,K,S,V=e.shape; B,T=x.shape
        b=(w[:,:,None]*i)[:,None].expand(-1,B,-1,-1)
        loss=torch.zeros(R); per=[]
        for t in range(T):
            pred=torch.einsum('rbks,rksv->rbv',b,e).clamp_min(1e-12)
            kl=(target[None,:,t]*(target[None,:,t].clamp_min(1e-12).log()-pred.log())).sum(-1)
            per.append(kl.mean(1)); loss+=kl.mean(1)/T
            # Prediction at t is BEFORE observing x_t. Emission then transition.
            likelihood=e[:,:,:,x[:,t]].permute(0,3,1,2)
            b=b*likelihood; b=b/b.sum((-1,-2),keepdim=True).clamp_min(1e-20)
            b=torch.einsum('rbks,rksu->rbku',b,a)
        return (loss,torch.stack(per,-1)) if detail else loss
    def candidate(self,r):
        w,i,a,e=[v.detach().cpu().numpy()[r].astype(float) for v in self.params()]
        # Float32 roundoff is removed in exported stochastic rows.
        norm=lambda x:(x/x.sum(-1,keepdims=True)).tolist()
        return {'schema_version':1,'weights':norm(w),'initial':norm(i),'transition':norm(a),'emission':norm(e)}

def run(case,steps,restarts,seed):
    started=time.monotonic(); folder=ROOT/case
    shape=json.loads((folder/'task.json').read_text())['shape']
    # Read all permitted evidence, but do not instantiate or call the transformer.
    checkpoint=torch.load(folder/'checkpoint.pt',map_location='cpu',weights_only=True)
    acts=np.load(folder/'activations.npy')
    xn=np.load(folder/'tokens.npy'); pn=np.load(folder/'probabilities.npy')
    pn=pn/pn.sum(-1,keepdims=True)
    rng=np.random.default_rng(seed); order=rng.permutation(len(xn)); train=order[:96]; valid=order[96:]
    x=torch.tensor(xn,dtype=torch.long); p=torch.tensor(pn,dtype=torch.float32)
    model=Fits(xn[train],pn[train],shape['components'],shape['states'],restarts,seed)
    opt=torch.optim.Adam(model.parameters(),lr=.045)
    best_score=float('inf'); best_candidate=None; history=[]; best_step=None; best_restart=None
    for step in range(steps+1):
        if step%100==0 or step==steps:
            with torch.no_grad():
                vl=model.score(x[valid],p[valid]); tl=model.score(x[train],p[train])
                ri=int(vl.argmin()); value=float(vl[ri])
                history.append({'step':step,'train_kl':tl.tolist(),'validation_kl':vl.tolist()})
                if value<best_score:
                    best_score=value; best_candidate=model.candidate(ri); best_step=step; best_restart=ri
                print(case,step,'train',float(tl[ri]),'valid',value,'seconds',round(time.monotonic()-started,2),flush=True)
        if step==steps: break
        opt.zero_grad(set_to_none=True)
        loss=model.score(x[train],p[train]); loss.sum().backward()
        torch.nn.utils.clip_grad_norm_(model.parameters(),10.)
        opt.step()
        if step in [steps//2,steps*4//5]:
            for group in opt.param_groups: group['lr']*=.4
    # Fixed all-data refit after restart/early-stopping selection.
    final=Fits(xn,pn,shape['components'],shape['states'],1,seed+100)
    with torch.no_grad():
        for attr,key in [('wl','weights'),('il','initial'),('al','transition'),('el','emission')]:
            getattr(final,attr).copy_(torch.tensor(np.log(np.maximum(best_candidate[key],1e-12)),dtype=torch.float32)[None])
    refit_opt=torch.optim.Adam(final.parameters(),lr=.005)
    for _ in range(400):
        refit_opt.zero_grad(set_to_none=True)
        final.score(x,p).sum().backward()
        torch.nn.utils.clip_grad_norm_(final.parameters(),10.)
        refit_opt.step()
    best_candidate=final.candidate(0)
    with torch.no_grad():
        fullkl,perposition=final.score(x,p,detail=True)
        refit_train=float(final.score(x[train],p[train])[0]); refit_old_valid=float(final.score(x[valid],p[valid])[0])
    baseline=p[train].mean((0,1))
    baseline_kl=float((p[valid]*(p[valid].log()-baseline.log())).sum(-1).mean())
    (folder/'candidate.json').write_text(json.dumps(best_candidate,indent=2)+'\n')
    return {'case':case,'final_refit':{'steps':400,'lr':.005,'histories':128,'full_data_kl':float(fullkl[0]),'per_position_full_data_kl':perposition[0].tolist(),'original_train_kl':refit_train,'original_validation_kl_now_in_sample':refit_old_valid,'validation_is_no_longer_held_out':True},'unigram_baseline_validation_kl':baseline_kl,'seconds':time.monotonic()-started,'selected_step':best_step,'selected_restart':best_restart,'selected_validation_kl_nats_per_token':best_score,'history':history,'train_sequence_indices':train.tolist(),'validation_sequence_indices':valid.tolist(),'architecture':checkpoint['architecture'],'activation_shape':list(acts.shape),'activation_use':'loaded and shape checked; not used in fitting','checkpoint_use':'architecture and state dictionary inspected; no transformer instantiated or evaluated','shape':shape}

def main():
    parser=argparse.ArgumentParser();parser.add_argument('--steps',type=int,default=1800);parser.add_argument('--restarts',type=int,default=12);parser.add_argument('--cases',nargs='+',default=['demo','pilot']);args=parser.parse_args()
    torch.set_num_threads(1); torch.manual_seed(806)
    started=time.monotonic(); results=[run(c,args.steps,args.restarts,806+j) for j,c in enumerate(args.cases)]
    report={'algorithm':'unsupervised multistart predictive distillation into a component-fixed node-emitting mixture HMM','objective':'mean forward KL(cached transformer next-token distribution || mixture-HMM filter next-token distribution), averaged over train histories and all 24 prediction positions','selection':'lowest held-out forward KL across restarts and checkpoints every 100 steps; then fixed 400-step all-data refit; no scorer or ground truth access','hyperparameters':{'steps':args.steps,'restarts':args.restarts,'optimizer':'Adam','initial_lr':.045,'lr_decays':{'fractions':[.5,.8],'factor':.4},'gradient_clip_norm':10,'train_histories':96,'validation_histories':32,'seed':806,'dtype':'float32','threads':1,'final_all_data_refit_steps':400,'final_all_data_refit_lr':.005,'initialization':'three quarters late-prediction sequence kmeans + lognormal emission perturbation; remainder Dirichlet emissions; random initial and transition logits'},'budget_per_case':{'cached_generation_forward_calls':24,'cached_token_queries':3072,'cached_prefix_tokens_processed':38400,'additional_transformer_forward_calls':0},'fitting_seconds':time.monotonic()-started,'cases':results,'limitations':['No ground truth or scoring feedback was available. Recovery is not demonstrated.','Distillation targets an approximate transformer, so low predictive KL does not certify identification of its generating HMM.','Finite histories, length 24, state-label permutations, weakly visited states and nonidentifiability can affect parameter recovery.','Activations and raw weights were read but not used beyond architecture inspection; optimization uses only cached tokens and probabilities.','Selection reuses a held-out set across restarts and checkpoints, so pre-refit validation scores have selection bias. Final refit includes all histories and has no held-out score.']}
    (ROOT/'report.json').write_text(json.dumps(report,indent=2)+'\n')
if __name__=='__main__': main()
